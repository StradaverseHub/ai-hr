CREATE TABLE public.contract_shared_email_document_step(
 tenant_id uuid NOT NULL REFERENCES public.tenant(id),case_id uuid NOT NULL,workflow_run_id uuid NOT NULL,workflow_step_id text NOT NULL,
 email_template_id uuid NOT NULL,email_revision integer NOT NULL,subject_person_id uuid NOT NULL,decision_revision integer NOT NULL,
 phase text NOT NULL CHECK(phase IN('outcome','declined')),documents jsonb NOT NULL DEFAULT '{}'::jsonb CHECK(jsonb_typeof(documents)='object'),
 prepared_at timestamptz NOT NULL DEFAULT now(),PRIMARY KEY(tenant_id,workflow_run_id,workflow_step_id),
 FOREIGN KEY(tenant_id,case_id) REFERENCES public.contract_shared_case(tenant_id,case_id),
 FOREIGN KEY(tenant_id,workflow_run_id) REFERENCES public.automation_run(tenant_id,id),
 FOREIGN KEY(tenant_id,email_template_id,email_revision) REFERENCES public.automation_email_template_version(tenant_id,template_id,revision)
);
ALTER TABLE public.contract_shared_email_document_step ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.contract_shared_email_document_step FORCE ROW LEVEL SECURITY;
CREATE POLICY contract_shared_email_document_step_owner ON public.contract_shared_email_document_step FOR ALL TO ai_hr_owner USING(app.acting_as_owner()) WITH CHECK(app.acting_as_owner());
REVOKE ALL ON public.contract_shared_email_document_step FROM PUBLIC,ai_hr_app;

CREATE FUNCTION app.contract_shared_email_document_step_immutable() RETURNS trigger LANGUAGE plpgsql SET search_path=pg_catalog,public,pg_temp AS $$
BEGIN
 IF (to_jsonb(NEW)-'documents') IS DISTINCT FROM (to_jsonb(OLD)-'documents')
 OR EXISTS(SELECT 1 FROM jsonb_each(OLD.documents) old WHERE NEW.documents->old.key IS DISTINCT FROM old.value) THEN RAISE EXCEPTION 'Prepared email references and document bindings are immutable' USING ERRCODE='23514';END IF;
 RETURN NEW;
END $$;
REVOKE ALL ON FUNCTION app.contract_shared_email_document_step_immutable() FROM PUBLIC,ai_hr_app;
CREATE TRIGGER contract_shared_email_document_step_immutable BEFORE UPDATE ON public.contract_shared_email_document_step FOR EACH ROW EXECUTE FUNCTION app.contract_shared_email_document_step_immutable();

CREATE FUNCTION app.contract_shared_email_documents_prepare(p_case uuid,p_run uuid,p_step text,p_email uuid,p_revision integer) RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $$
DECLARE pin public.contract_shared_email_document_step;f public.contract_shared_case;c public.contract_case;definition jsonb;expected text;document_ids jsonb;person uuid;
BEGIN
 IF app.contract_shared_effect_held(p_case,p_run,p_step,'sendEmail') IS NOT TRUE THEN RAISE EXCEPTION 'Protected Send email step required' USING ERRCODE='42501';END IF;
 PERFORM app.contract_shared_lock_source(p_case);
 SELECT * INTO f FROM public.contract_shared_case WHERE tenant_id=app.current_tenant_id() AND case_id=p_case;
 SELECT * INTO c FROM public.contract_case WHERE tenant_id=f.tenant_id AND id=p_case;
 SELECT employment.person_id INTO person FROM public.employment employment WHERE tenant_id=c.tenant_id AND id=c.employment_id;
 SELECT * INTO pin FROM public.contract_shared_email_document_step WHERE tenant_id=f.tenant_id AND workflow_run_id=p_run AND workflow_step_id=p_step FOR UPDATE;
 IF pin.workflow_run_id IS NULL THEN
  SELECT step->'params'->>'emailTemplateId' INTO expected FROM public.automation_run run JOIN public.automation_workflow_version workflow ON workflow.tenant_id=run.tenant_id AND workflow.workflow_id=run.workflow_id AND workflow.revision=run.workflow_revision
  CROSS JOIN LATERAL jsonb_array_elements(workflow.definition->'steps') step WHERE run.tenant_id=f.tenant_id AND run.id=p_run AND step->>'id'=p_step AND step->>'type'='sendEmail';
  IF expected IS DISTINCT FROM p_email::text THEN RAISE EXCEPTION 'The frozen email step changed' USING ERRCODE='40001';END IF;
  SELECT to_jsonb(version) INTO definition FROM public.automation_email_template_version version WHERE tenant_id=f.tenant_id AND template_id=p_email AND revision=p_revision AND module='contract_staff' AND active AND NOT retired
  AND revision=(SELECT max(latest.revision) FROM public.automation_email_template_version latest WHERE latest.tenant_id=f.tenant_id AND latest.template_id=p_email);
  IF definition IS NULL THEN RAISE EXCEPTION 'Review the current email definition before preparing documents' USING ERRCODE='40001';END IF;
  INSERT INTO public.contract_shared_email_document_step(tenant_id,case_id,workflow_run_id,workflow_step_id,email_template_id,email_revision,subject_person_id,decision_revision,phase)
  VALUES(f.tenant_id,p_case,p_run,p_step,p_email,p_revision,person,coalesce((f.data->'decision'->>'revision')::integer,0),CASE WHEN f.data->'declined'='null'::jsonb THEN 'outcome' ELSE 'declined' END) RETURNING * INTO pin;
 ELSE
  IF pin.case_id<>p_case OR pin.subject_person_id<>person OR pin.decision_revision<>coalesce((f.data->'decision'->>'revision')::integer,0) OR pin.phase<>CASE WHEN f.data->'declined'='null'::jsonb THEN 'outcome' ELSE 'declined' END THEN RAISE EXCEPTION 'The prepared email source changed' USING ERRCODE='40001';END IF;
  SELECT to_jsonb(version) INTO definition FROM public.automation_email_template_version version WHERE tenant_id=pin.tenant_id AND template_id=pin.email_template_id AND revision=pin.email_revision AND module='contract_staff';
 END IF;
 document_ids:=coalesce(nullif(definition->'document_ids','null'::jsonb),'[]'::jsonb);
 IF jsonb_typeof(document_ids)<>'array' OR jsonb_array_length(document_ids)>100 OR EXISTS(SELECT 1 FROM jsonb_array_elements(document_ids) id WHERE jsonb_typeof(id)<>'string' OR id#>>'{}' !~ '^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$') THEN RAISE EXCEPTION 'The email document list is invalid' USING ERRCODE='22023';END IF;
 RETURN jsonb_build_object('reference',jsonb_build_object('id',pin.email_template_id,'revision',pin.email_revision),'template',definition,'documentTemplateIds',document_ids,'documents',pin.documents);
END $$;
REVOKE ALL ON FUNCTION app.contract_shared_email_documents_prepare(uuid,uuid,text,uuid,integer) FROM PUBLIC,ai_hr_app;

CREATE FUNCTION app.contract_shared_email_document_reuse(p_case uuid,p_run uuid,p_step text,p_template uuid) RETURNS uuid
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $$
DECLARE pin public.contract_shared_email_document_step;run public.automation_run;event public.contract_shared_event;result uuid;step_order bigint;
BEGIN
 IF app.contract_shared_effect_held(p_case,p_run,p_step,'sendEmail') IS NOT TRUE THEN RAISE EXCEPTION 'Protected Send email step required' USING ERRCODE='42501';END IF;
 SELECT * INTO pin FROM public.contract_shared_email_document_step WHERE tenant_id=app.current_tenant_id() AND workflow_run_id=p_run AND workflow_step_id=p_step AND case_id=p_case;
 IF pin.workflow_run_id IS NULL OR NOT EXISTS(SELECT 1 FROM public.automation_email_template_version version WHERE tenant_id=pin.tenant_id AND template_id=pin.email_template_id AND revision=pin.email_revision AND coalesce(to_jsonb(version)->'document_ids','[]'::jsonb) ? p_template::text) THEN RAISE EXCEPTION 'This email definition does not reference the document' USING ERRCODE='42501';END IF;
 SELECT * INTO run FROM public.automation_run WHERE tenant_id=pin.tenant_id AND id=p_run;
 SELECT * INTO event FROM public.contract_shared_event WHERE tenant_id=pin.tenant_id AND id::text=current_setting('app.contract_event',true);
 SELECT id INTO result FROM public.contract_shared_letter WHERE tenant_id=pin.tenant_id AND case_id=p_case AND decision_revision=pin.decision_revision AND workflow_run_id=p_run AND workflow_step_id=p_step AND data->>'templateId'=p_template::text AND purpose='email';
 IF result IS NOT NULL THEN RETURN result;END IF;
 SELECT n INTO step_order FROM public.automation_workflow_version version CROSS JOIN LATERAL jsonb_array_elements(version.definition->'steps') WITH ORDINALITY steps(step,n)
 WHERE version.tenant_id=run.tenant_id AND version.workflow_id=run.workflow_id AND version.revision=run.workflow_revision AND step->>'id'=p_step;
 SELECT letter.id INTO result FROM public.contract_shared_letter letter JOIN public.automation_workflow_version version ON version.tenant_id=run.tenant_id AND version.workflow_id=run.workflow_id AND version.revision=run.workflow_revision
 CROSS JOIN LATERAL jsonb_array_elements(version.definition->'steps') WITH ORDINALITY steps(step,n)
 WHERE letter.tenant_id=pin.tenant_id AND letter.case_id=p_case AND letter.decision_revision=pin.decision_revision AND letter.purpose=pin.phase AND letter.workflow_run_id=p_run
 AND letter.data->>'templateId'=p_template::text AND letter.workflow_step_id=step->>'id' AND step->>'type'='generateDocument' AND n<step_order AND letter.data->>'state' NOT IN('declined','withdrawn') ORDER BY n DESC LIMIT 1;
 IF result IS NOT NULL THEN RETURN result;END IF;
 IF event.trigger_event IN('letterReleased','letterActionNeeded','letterSentBack') AND run.event_key=event.trigger_event||'|'||event.occurrence THEN
  SELECT letter.id INTO result FROM public.contract_shared_letter letter WHERE letter.tenant_id=pin.tenant_id AND letter.case_id=p_case AND letter.decision_revision=pin.decision_revision AND letter.purpose=pin.phase
  AND letter.id::text=event.data->>'letterId' AND letter.data->>'templateId'=p_template::text AND letter.data->>'state' NOT IN('declined','withdrawn');
 END IF;
 RETURN result;
END $$;
REVOKE ALL ON FUNCTION app.contract_shared_email_document_reuse(uuid,uuid,text,uuid) FROM PUBLIC,ai_hr_app;

CREATE FUNCTION app.contract_shared_email_document_bind(p_case uuid,p_run uuid,p_step text,p_template uuid,p_letter uuid) RETURNS void
LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $$
DECLARE pin public.contract_shared_email_document_step;actual uuid;
BEGIN
 actual:=app.contract_shared_email_document_reuse(p_case,p_run,p_step,p_template);
 IF actual IS DISTINCT FROM p_letter THEN RAISE EXCEPTION 'The canonical email document receipt differs' USING ERRCODE='40001';END IF;
 SELECT * INTO pin FROM public.contract_shared_email_document_step WHERE tenant_id=app.current_tenant_id() AND workflow_run_id=p_run AND workflow_step_id=p_step FOR UPDATE;
 IF pin.documents ? p_template::text THEN IF pin.documents->>p_template::text IS DISTINCT FROM p_letter::text THEN RAISE EXCEPTION 'The email document was already bound' USING ERRCODE='40001';END IF;RETURN;END IF;
 UPDATE public.contract_shared_email_document_step SET documents=jsonb_set(documents,ARRAY[p_template::text],to_jsonb(p_letter::text)) WHERE tenant_id=pin.tenant_id AND workflow_run_id=p_run AND workflow_step_id=p_step;
END $$;
REVOKE ALL ON FUNCTION app.contract_shared_email_document_bind(uuid,uuid,text,uuid,uuid) FROM PUBLIC,ai_hr_app;

CREATE FUNCTION app.contract_shared_email_document_readiness(p_letter uuid) RETURNS jsonb
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $$
DECLARE letter public.contract_shared_letter;document jsonb;subject_person uuid;
BEGIN
 -- Private helper; only the held-step owner and the authenticated canonical progression trigger call it.
 -- No application EXECUTE grant. A human-origin trigger retains its real actor, without impersonating owner.
 SELECT * INTO letter FROM public.contract_shared_letter WHERE tenant_id=app.current_tenant_id() AND id=p_letter;
 SELECT data INTO document FROM public.document_signing_document WHERE tenant_id=letter.tenant_id AND id=letter.signing_document_id;
 SELECT employment.person_id INTO subject_person FROM public.contract_case c JOIN public.employment employment ON employment.tenant_id=c.tenant_id AND employment.id=c.employment_id WHERE c.tenant_id=letter.tenant_id AND c.id=letter.case_id;
 IF letter.id IS NULL OR document IS NULL OR subject_person::text IS DISTINCT FROM letter.data->>'subjectPersonId' THEN RETURN jsonb_build_object('ready',false,'reason','source_changed');END IF;
 IF letter.data->>'state' IN('declined','withdrawn') THEN RETURN jsonb_build_object('ready',false,'reason','stopped');END IF;
 IF app.document_signing_returned(document) THEN RETURN jsonb_build_object('ready',false,'reason','returned');END IF;
 IF letter.data->>'state'='released' THEN RETURN jsonb_build_object('ready',true,'reason','released');END IF;
 IF EXISTS(SELECT 1 FROM app.document_signing_pending_steps(document) step WHERE step->>'role'='employee') THEN
  IF EXISTS(SELECT 1 FROM app.document_signing_pending_steps(document) step JOIN public.app_user account ON account.tenant_id=letter.tenant_id AND account.id::text=step->>'holderId' AND account.person_id::text=step->>'holderPersonId'
    WHERE step->>'role'='employee' AND account.person_id=subject_person AND app.contract_verified_human(account.id)) THEN RETURN jsonb_build_object('ready',true,'reason','employee');END IF;
  RETURN jsonb_build_object('ready',false,'reason','employee_account');
 END IF;
 RETURN jsonb_build_object('ready',false,'reason','hr_steps');
END $$;
REVOKE ALL ON FUNCTION app.contract_shared_email_document_readiness(uuid) FROM PUBLIC,ai_hr_app;

-- 1. Canonical document progression/account-binding derives ready transitions; no public enqueue endpoint.
-- 2. Enqueue internal emailDocumentReady with versions=[], exact waiting run/step, document ID/revision and
--    pinned email reference. Native event admission freezes the real acting UID/PID and actual source person.
-- 3. Worker proof binds tenant/job capability/root event, original run actor PID, pinned source person,
--    case decision revision/phase and current case company. It refuses source-person/company changes.
-- 4. A current accepted period resumes its already-authorised waiting run normally. If only the period
--    changed after posting, historical proof may resume only its pinned Send email step; no new approval,
--    posting, Offboarding or Personnel/Payroll effect is admitted by that historical exception.
-- 5. Shared runtime retryStep=true reruns the SAME email step with its immutable pin. OneHR uses the same
--    run+step key; duplicate events cannot queue another email. No second cursor/state engine is added.
