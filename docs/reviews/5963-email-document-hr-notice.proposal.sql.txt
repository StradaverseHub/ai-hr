CREATE FUNCTION app.contract_shared_email_hr_notice_source(p_letter uuid,p_revision integer) RETURNS jsonb
LANGUAGE sql STABLE SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $$
 SELECT jsonb_build_object('letterId',l.id,'pinId',pin.id,'companyId',c.company_id,'subjectPersonId',e.person_id,'signingRevision',d.revision,
 'recipients',coalesce((SELECT jsonb_agg(jsonb_build_object('userId',u.id,'personId',u.person_id,'stepId',step->>'id') ORDER BY u.id,step->>'id') FROM app.document_signing_pending_steps(d.data) step
 JOIN public.app_user u ON u.tenant_id=l.tenant_id AND u.id::text=step->>'holderId' AND u.person_id::text=step->>'holderPersonId'
 WHERE u.person_id<>e.person_id AND app.contract_verified_human(u.id)), '[]'::jsonb),
 'returned',app.document_signing_returned(d.data),'ownerUserId',app.document_signing_owner(d.data))
 FROM public.contract_shared_letter l JOIN public.document_signing_document d ON d.tenant_id=l.tenant_id AND d.id=l.signing_document_id AND d.subject_type='contract' AND d.subject_id=l.id
 JOIN public.contract_case c ON c.tenant_id=l.tenant_id AND c.id=l.case_id
 JOIN public.employment e ON e.tenant_id=c.tenant_id AND e.id=c.employment_id AND e.company_id=c.company_id AND e.person_id::text=l.data->>'subjectPersonId' AND e.voided_at IS NULL
 JOIN public.contract_shared_email_document_step pin ON pin.tenant_id=l.tenant_id AND pin.case_id=l.case_id AND pin.company_id=c.company_id AND pin.subject_person_id=e.person_id AND pin.documents->>(l.data->>'templateId')=l.id::text
 WHERE l.tenant_id=app.current_tenant_id() AND l.id=p_letter AND l.purpose='email' AND l.data->>'state'='routing' AND d.revision=p_revision AND app.contract_company_entitled(c.company_id)
 AND app.contract_shared_email_document_current(l.id)
$$;
REVOKE ALL ON FUNCTION app.contract_shared_email_hr_notice_source(uuid,integer) FROM PUBLIC,ai_hr_app;
